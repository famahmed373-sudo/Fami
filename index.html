function setupRealtime() {
if (!uid()) return;
console.log("Setting up realtime for user:", uid());
if (realtimeChannel) sb.removeChannel(realtimeChannel);

realtimeChannel = sb.channel(`user:${uid()}`);
realtimeChannel.on('broadcast', { event: 'new_notification' }, ({ payload }) => {
console.log('Received broadcast notification:', payload);
showToast(payload.message, payload.type === 'error', payload.duration || 3500);
});

// Subscribe to changes that might affect the current user's view
sb.channel('payments_changes')
.on('postgres_changes', {
event: '*',
schema: 'public',
table: 'payments',
}, (payload) => {
console.log('Payment change detected:', payload);
refreshAll(); // Simplest approach: refresh all data
})
.subscribe();

sb.channel('shops_changes')
.on('postgres_changes', {
event: '*',
schema: 'public',
table: 'shops',
}, (payload) => {
console.log('Shop change detected:', payload);
refreshAll();
})
.subscribe();

// Subscribe to user role changes if admin
if (state.profile?.role === 'admin') {
sb.channel('profiles_changes')
.on('postgres_changes', {
event: '*',
schema: 'public',
table: 'profiles',
}, (payload) => {
console.log('Profile change detected:', payload);
refreshAll();
})
.subscribe();
}

realtimeChannel.subscribe();
console.log("Realtime channels subscribed.");
}

function cleanupRealtime() {
console.log("Cleaning up realtime channels.");
if (realtimeChannel) {
sb.removeChannel(realtimeChannel);
realtimeChannel = null;
}
sb.removeAllChannels(); // Ensure all channels are unsubscribed
}

function showToast(msg, isError = false, duration = 3200) {
const t = $('toast'), m = $('toastMsg');
m.textContent = msg;
t.className = 'toast' + (isError ? ' error' : '') + ' show';
clearTimeout(toastTimer);
toastTimer = setTimeout(() => t.classList.remove('show'), duration);
}

function setAuthMode(mode) {
state.authMode = mode;
$('loginTab').classList.toggle('active', mode === 'login');
$('signupTab').classList.toggle('active', mode === 'signup');
$('nameField').style.display = mode === 'signup' ? 'grid' : 'none';

function authMessage(msg, isError = false) {
$('authMsg').textContent = msg;
$('authMsg').className = 'notice' + (isError ? ' dangerNotice' : '');
$('authMsg').classList.remove('hidden');
}

async function submitAuth() {
const email = $('authEmail').value.trim();
const password = $('authPassword').value;
const name = $('authName').value.trim();

if (!email || password.length < 6) return authMessage('Enter a valid email and a password of at least 6 characters.', true);

$('authBtn').disabled = true;
try {
if (state.authMode === 'signup') {
const { data, error } = await sb.auth.signUp({
email,
password,
options: {
data: { full_name: name, role: 'viewer' } // Default role for new signups
}
});
if (error) throw error;
// After signup, user needs to confirm email (if enabled) and then sign in.
// Profile creation is handled by the trigger on auth.users.
authMessage('Account created. Please check your email for verification instructions.', false);
setAuthMode('login'); // Switch to login mode after signup
} else { // Sign in
const { error } = await sb.auth.signInWithPassword({ email, password });
if (error) throw error;
// onAuthStateChange listener will handle moving to startApp()
}
} catch (e) {
authMessage(e.message || 'Authentication failed.', true);
} finally {
$('authBtn').disabled = false;
}
}

async function logout() {
cleanupRealtime();
try {
await sb.auth.signOut();
// onAuthStateChange listener will handle moving to showLogin()
} catch (error) {
console.error("Error during logout:", error);
showToast("An error occurred during logout.", true);
}
}

async function boot() {
console.log("Booting FAMI Rent...");
try {
const { data } = await sb.auth.getSession();
state.session = data.session;

sb.auth.onAuthStateChange((event, session) => {
state.session = session;
if (session) {
console.log("User session established:", session.user.id);
startApp();
} else {
console.log("User logged out.");
showLogin();
cleanupRealtime();
}
});

if (state.session) {
console.log("Session found, starting app...");
startApp();
} else {
console.log("No session found, showing login.");
showLogin();
}
} catch (error) {
console.error("Supabase authentication error during boot:", error);
showToast("Authentication service unavailable. Please try again later.", true);
showLogin();
}
}

async function startApp() {
$('loginView').classList.add('hidden');
$('app').classList.remove('hidden');
state.loading = true;
renderLoadingStates(true); // Show skeletons while loading

try {
await loadProfile();
applyPermissions();
await refreshAll();
setupRealtime(); // Setup realtime subscriptions after app is ready
console.log("App started successfully.");
} catch (error) {
console.error("Error during app startup:", error);
showToast(`Failed to load application data: ${error.message}`, true);
logout(); // Force logout if critical data fails to load
} finally {
state.loading = false;
renderLoadingStates(false); // Hide skeletons
}
}

// Calculates outstanding amount for a specific shop for a specific month using RPC
async function getShopBalanceForMonth(shopId, month) {
if (!shopId || !month) return 0;
try {
const { data, error } = await sb.rpc('calculate_rent_due_for_month', { p_shop_id: shopId, p_month: month });
if (error) {
console.error(`Error calculating balance for ${month}:`, error);
return 0;
}
return data || 0;
} catch (error) {
console.error("RPC Error getting shop balance:", error);
return 0;
}
}

// Calculates total outstanding for a shop across all months
async function getTotalShopOutstanding(shopId) {
if (!shopId) return { owed: 0, unpaidMonthsCount: 0, totalPaid: 0, totalDue: 0 };

let totalOwed = 0;
let unpaidMonthsCount = 0;
let totalPaid = 0;
let totalDue = 0;

const shop = state.shops.find(s => s.id === shopId);
if (!shop) return { owed: 0, unpaidMonthsCount: 0, totalPaid: 0, totalDue: 0 };

const registrationMonth = shop.registration_date.slice(0, 7);
const currentMonthStr = currentMonth();

// Iterate through months from registration to current
let month = new Date(registrationMonth + '-01T00:00:00Z');
const endMonth = new Date(currentMonthStr + '-01T00:00:00Z');

while (month <= endMonth) {
const monthKey = `${month.getFullYear()}-${String(month.getMonth() + 1).padStart(2, '0')}`;
const balance = await getShopBalanceForMonth(shopId, monthKey);
const rentForMonth = Number(shop.monthly_rent || 0);
const paidForMonth = state.payments.filter(p => p.shop_id === shopId && p.payment_month.startsWith(monthKey) && p.status === 'posted').reduce((a,p) => a + Number(p.amount), 0);

totalDue += rentForMonth;
totalPaid += paidForMonth;

if (balance > 0) {
totalOwed += balance;
unpaidMonthsCount++;
}
month.setMonth(month.getMonth() + 1);
}
return { owed: totalOwed, unpaidMonthsCount, totalPaid, totalDue };
}

// Add Payment Function (Simplified)
async function addPayment() {
if (!['admin', 'manager', 'payment_officer'].includes(state.profile?.role)) {
return showToast('You do not have permission to record payments.', true);
}

const shop_id = $('payShop').value;
const payment_month = $('payMonth').value;
const amount = Number($('payAmount').value);
const payment_date = $('payDate').value;
const method = $('payMethod').value;
const reference = $('payRef').value.trim();
const note = $('payNote').value.trim();

if (!shop_id || !payment_month || !amount || amount <= 0 || !payment_date) {
return showToast('Complete all required payment fields.', true);
}

const shop = state.shops.find(s => s.id === shop_id);
const currentTotalPaid = state.payments.filter(p => p.shop_id === shop_id && p.payment_month.startsWith(payment_month) && p.status === 'posted').reduce((a,p) => a + Number(p.amount), 0);
const maxDueForMonth = Number(shop?.monthly_rent || 0);

if (currentTotalPaid >= maxDueForMonth) {
return showToast('This shop is already fully paid for that month.', true);
}
if (amount > maxDueForMonth - currentTotalPaid) {
if (!confirm(`This payment (${fmt(amount)}) exceeds the remaining balance for ${monthLabel(payment_month)}. The remaining balance is ${fmt(maxDueForMonth - currentTotalPaid)}. Continue with overpayment?`)) {
return;
}
}

try {
const { data, error } = await sb.from('payments').insert({
shop_id,
payment_month: monthDate(payment_month),
amount,
payment_date,
method,
reference,
note,
created_by: uid()
}).select().single();

if (error) throw error;

await audit('RECORD PAYMENT', 'payment', data.id, { shop_id, amount, payment_month, method, reference });
showToast('Payment recorded successfully.');

// Clear form fields
$('payShop').value = '';
$('payMonth').value = currentMonth();
$('payAmount').value = '';
$('payDate').value = today();
$('payMethod').value = 'Cash';
$('payRef').value = '';
$('payNote').value = '';

await refreshAll(); // Refresh data
} catch (error) {
console.error("Error adding payment:", error);
showToast(error.message || 'Could not record payment.', true);
}
}

// Reports Render (Simplified)
function renderReports() {
const year = parseInt($('reportYear').value || new Date().getFullYear());
const rows = [];
let totalExpected = 0;
let totalCollected = 0;

for (let i = 0; i < 12; i++) {
const monthNum = (i + 1).toString().padStart(2, '0');
const monthKey = `${year}-${monthNum}`;
const monthLabelText = monthLabel(monthKey);

let expected = 0;
let collected = 0;

// Calculate expected rent for this month
state.shops.filter(s => s.status === 'active' && s.registration_date.slice(0, 7) <= monthKey).forEach(s => {
expected += Number(s.monthly_rent || 0);
});

// Calculate collected rent for this month
collected = state.payments.filter(p =>
p.status === 'posted' && p.payment_month.startsWith(monthKey)
).reduce((sum, p) => sum + Number(p.amount), 0);

totalExpected += expected;
totalCollected += collected;

const outstanding = Math.max(0, expected - collected);
const collectionRate = expected ? Math.round((collected / expected) * 100) : 0;

rows.push(`<tr>
<td><b>${monthLabelText}</b></td>
<td>${fmt(expected)}</td>
<td>${fmt(collected)}</td>
<td>${fmt(outstanding)}</td>
<td><b>${collectionRate}%</b><div class="progress" style="margin-top:5px"><i style="width:${collectionRate}%"></i></div></td>
</tr>`);
}

// Add summary row
const totalOutstanding = Math.max(0, totalExpected - totalCollected);
const totalCollectionRate = totalExpected ? Math.round((totalCollected / totalExpected) * 100) : 0;
rows.push(`<tr>
<th><b>Total</b></th>
<th><b>${fmt(totalExpected)}</b></th>
<th><b>${fmt(totalCollected)}</b></th>
<th><b>${fmt(totalOutstanding)}</b></th>
<th><b>${totalCollectionRate}%</b></th>
</tr>`);

$('reportCards').innerHTML = `
<div class="card reportCard"><small>Expected Total</small><b>${fmt(totalExpected)}</b></div>
<div class="card reportCard"><small>Collected Total</small><b>${fmt(totalCollected)}</b></div>
<div class="card reportCard"><small>Outstanding Total</small><b>${fmt(totalOutstanding)}</b></div>
<div class="card reportCard"><small>Overall Collection Rate</small><b>${totalCollectionRate}%</b></div>

async function saveShop() {
const id = state.editingShop;
const payload = {
shop_code: $('mCode').value.trim(),
tenant_name: $('mTenant').value.trim(),
phone: $('mPhone').value.trim(),
business_name: $('mBusiness').value.trim(),
monthly_rent: Number($('mRent').value),
registration_date: $('mReg').value,
notes: $('mNotes').value.trim()
};

if (!payload.shop_code || !payload.tenant_name || !payload.monthly_rent || !payload.registration_date) {
return showToast('Complete required shop fields.', true);
}

try {
let res;
if (id) { // Editing
res = await sb.from('shops').update(payload).eq('id', id);
await audit('UPDATE SHOP', 'shop', id, { shop_code: payload.shop_code });
} else { // Creating
payload.created_by = uid();
res = await sb.from('shops').insert(payload).select().single();
await audit('REGISTER SHOP', 'shop', res.data.id, { shop_code: payload.shop_code, tenant: payload.tenant_name });
}

if (res.error) throw res.error;

closeModal();
showToast(id ? 'Shop updated.' : 'Shop registered.');
await refreshAll();
} catch (error) {
console.error("Error saving shop:", error);
showToast(error.message || 'Could not save shop.', true);
}
}

// Toggle Shop Status
async function toggleShop(id) {
if (!['admin', 'manager'].includes(state.profile?.role)) return showToast('You do not have permission.', true);

const shop = state.shops.find(x => x.id === id);
if (!shop) return;
const nextStatus = shop.status === 'active' ? 'released' : 'active';
const actionText = nextStatus === 'active' ? 'Reactivate' : 'Release';

if (!confirm(`${actionText} shop ${shop.shop_code}?`)) return;

try {
const { error } = await sb.from('shops').update({ status: nextStatus }).eq('id', id);
if (error) throw error;
await audit(nextStatus === 'active' ? 'REACTIVATE SHOP' : 'RELEASE SHOP', 'shop', id, { shop_code: shop.shop_code });
showToast('Shop status updated.');
await refreshAll();
} catch (error) {
console.error("Error toggling shop status:", error);
showToast(error.message, true);
}
}

// Populate shop dropdown for payment form
function populatePaymentShop() {
const current = $('payShop')?.value;
const shops = state.shops.filter(s => s.status === 'active'); // Only show active shops for payment recording
$('payShop').innerHTML = '<option value="">Select shop...</option>' + shops.map(s =>
`<option value="${s.id}">${esc(s.shop_code)} — ${esc(s.tenant_name)}</option>`
).join('');
if(current) $('payShop').value = current;
$('payMonth').value = $('payMonth').value || currentMonth();
$('payDate').value = $('payDate').value || today();
}

function suggestPaymentAmount(){
const shop_id = $('payShop').value;
const payment_month = $('payMonth').value;
if (!shop_id || !payment_month) return;

const shop = state.shops.find(s => s.id === shop_id);
if (!shop) return;

// Get balance for the selected month
getShopBalanceForMonth(shop_id, payment_month).then(balance => {
$('payAmount').value = balance > 0 ? balance : Number(shop.monthly_rent);
}).catch(err => {
console.error("Error suggesting amount:", err);
$('payAmount').value = Number(shop.monthly_rent); // Fallback
});
}

// Change User Role
async function changeRole(userId, newRole) {
if (userId === uid() && newRole !== 'admin' && state.profile.role === 'admin') {
if (!confirm('You are about to change your own role from Admin. This may restrict your access. Continue?')) {
await refreshAll(); // Re-render to reset the select
return;
}
}
try {
const { error } = await sb.from('profiles').update({ role: newRole }).eq('id', userId);
if (error) throw error;
await audit('CHANGE USER ROLE', 'profile', userId, { new_role: newRole, old_role: state.profiles.find(p => p.id === userId)?.role });
showToast('User role updated.');
await loadProfile(); // Reload current user's profile to apply permission changes
applyPermissions();
await refreshAll(); // Refresh all data to reflect role changes across UI
} catch (e) {
console.error("Error changing role:", e);
showToast(e.message || 'Failed to update role.', true);
await refreshAll(); // Revert UI if update fails
}
}

// Delete User
async function deleteUser(userId) {
if (!confirm('Are you sure you want to delete this user? This action cannot be undone.')) return;
try {
// Use Supabase RPC to delete user and cascade profile deletion
const { error } = await sb.rpc('auth.delete_user', { id: userId });
if (error) throw error;
await audit('DELETE USER', 'profile', userId, {});
showToast('User deleted successfully.');
await refreshAll(); // Refresh user list
} catch (e) {
console.error("Error deleting user:", e);
showToast(e.message || 'Failed to delete user.', true);
}
}
